# Tower of Hanoi using DFS (State-Space Search)

from collections import deque

def hanoi_bfs(n):
    start = (0,) * n
    goal = (2,) * n

    queue = deque([start])
    parent = {start: None}
    moves = {}

    while queue:
        state = queue.popleft()

        if state == goal:
            break

        for i in range(n):
            source = state[i]

            # Check whether disk is on top
            if any(state[j] == source for j in range(i)):
                continue

            for dest in [2, 1, 0]:
                if dest == source:
                    continue

                # Check whether destination is legal
                if any(state[j] == dest for j in range(i)):
                    continue

                new_state = list(state)
                new_state[i] = dest
                new_state = tuple(new_state)

                if new_state not in parent:
                    parent[new_state] = state
                    moves[new_state] = (i + 1, source, dest)
                    queue.append(new_state)

    # Trace the path from goal to start
    path = []
    current = goal

    while current != start:
        path.append(moves[current])
        current = parent[current]

    path.reverse()

    rods = ['A', 'B', 'C']

    print("Initial State: All disks on A")
    print("Goal State: All disks on C")
    print("Minimum moves required:", len(path))

    for disk, source, dest in path:
        print("Move disk", disk, "from",
              rods[source], "to", rods[dest])

    print("Goal Reached!")

n = int(input("Enter number of disks: "))

if n > 0:
    hanoi_bfs(n)
else:
    print("Enter a positive number of disks.")

Problem (TSP), in which every city is visited exactly once and the salesman returns to the starting

# Local Beam Search for TSP

import random
cities = {
 'A': (0, 0),
 'B': (2, 3),
 'C': (5, 2),
 'D': (6, 6),
 'E': (1, 5)
}
def distance(a, b):
 x1, y1 = cities[a]
 x2, y2 = cities[b]
 return ((x1-x2)**2 + (y1-y2)**2)**0.5
def cost(tour):
 total = 0
 for i in range(len(tour)):
 total += distance(tour[i], tour[(i+1) % len(tour)])
 return total
# Local Beam Search
k = 3
beam = []
for i in range(k):
 tour = list(cities.keys())
 random.shuffle(tour)
 beam.append(tour)
for step in range(100):
 candidates = []
 for tour in beam:
 for i in range(len(tour)):
 for j in range(i+1, len(tour)):
 new_tour = tour[:]
 new_tour[i], new_tour[j] = new_tour[j], new_tour[i]
 candidates.append(new_tour)
 candidates.sort(key=cost)
 beam = candidates[:k]
best = min(beam, key=cost)
print("Best Tour:", " -> ".join(best + [best[0]]))
print("Total Distance:", round(cost(best), 2))

from collections import deque

# 0 = Free path, 1 = Obstacle
grid = [
    [0, 0, 0, 0],
    [1, 1, 0, 1],
    [0, 0, 0, 0],
    [0, 1, 1, 0]
]

start = (0, 0)
goal = (3, 3)

def bfs():
    q = deque([(start, [start])])
    visited = {start}

    while q:
        (r, c), path = q.popleft()

        if (r, c) == goal:
            return path

        for dr, dc in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
            nr, nc = r + dr, c + dc

            if (0 <= nr < 4 and 0 <= nc < 4 and
                grid[nr][nc] == 0 and (nr, nc) not in visited):

                visited.add((nr, nc))
                q.append(((nr, nc), path + [(nr, nc)]))

    return None

path = bfs()

print("Initial Grid:")
for row in grid:
    print(row)

if path:
    print("\nRobot Path:")
    for position in path:
        print(position)

    print("\nTotal Moves:", len(path) - 1)
    print("Goal Reached!")
else:
    print("No path found")

 return False

 for p, q in states:

# 8-Queens using Backtracking

N = 8
board = [-1] * N
def safe(r, c):
 for i in range(r):
 if board[i] == c or abs(board[i]-c) == abs(i-r):
 return False
 return True
def solve(r):
 if r == N:
 return True
 for c in range(N):
 if safe(r, c):
 board[r] = c
 if solve(r+1):
 return True
 board[r] = -1
 return False
solve(0)
for r in range(N):
 for c in range(N):
 print("Q" if board[r] == c else ".", end=" ")
